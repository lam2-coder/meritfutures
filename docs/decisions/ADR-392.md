### 11. What is owed

1. **THE GATE LEG, WHICH IS THE OTHER HALF OF [ADR-389](ADR-389.md) SECTION 12 ITEM 2.** [`gates.mjs`](../../scripts/corpus/gates.mjs) is read-only to all three rows this wave, so the predicate is written out here rather than landed. It belongs beside `CI-06l`'s existing `cronRows(body, CRON_SCHEDULED)` call as a fifth assertion, and it needs no new `CI-06` letter:

   ```js
   const CRON_SEVERITY = '## Severity if absent, by job';

   // The `### <job>` notes, keyed the way the scheduled table's first cell is keyed.
   function severityNotes(body) {
     const start = body.indexOf(CRON_SEVERITY);
     if (start === -1) throw new Error(`${CRON_DOC}: section not found: "${CRON_SEVERITY}"`);
     const after = body.slice(start + CRON_SEVERITY.length);
     const end = after.search(/\n## /);
     const notes = new Map();
     let job = null;
     let lines = [];
     for (const line of (end === -1 ? after : after.slice(0, end)).split('\n')) {
       const heading = /^### (.+)$/.exec(line);
       if (heading) {
         if (job !== null) notes.set(job, lines.join('\n').trim());
         job = normJob(heading[1]);
         lines = [];
         continue;
       }
       if (job !== null) lines.push(line);
     }
     if (job !== null) notes.set(job, lines.join('\n').trim());
     return notes;
   }

   // Assertion 5, both directions, beside the existing scheduled-set build.
   const notes = severityNotes(body);
   for (const job of scheduled)
     if (!notes.has(job)) findings.push(`${CRON_DOC}: scheduled job "${job}" has no severity note`);
   for (const [job, note] of notes) {
     if (!scheduled.has(job))
       findings.push(`${CRON_DOC}: severity note "${job}" has no scheduled row`);
     else if (note === '') findings.push(`${CRON_DOC}: severity note "${job}" is empty`);
   }
   ```
